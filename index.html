Supabase Database Table Setup (1-Click SQL)

Run this in your Supabase Dashboard → SQL Editor to initialize the centralized table and enable real-time replication: 

-- 1. Create centralized table for Waytone ERP
CREATE TABLE IF NOT EXISTS waytone_erp_state (
  id TEXT PRIMARY KEY DEFAULT 'central_database',
  data JSONB NOT NULL,
  updated_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL,
  updated_by TEXT DEFAULT 'System'
);

-- 2. Seed initial empty record
INSERT INTO waytone_erp_state (id, data, updated_by)
VALUES ('central_database', '{}'::jsonb, 'Initial Setup')
ON CONFLICT (id) DO NOTHING;

-- 3. Enable Row Level Security (RLS) & permit anon access
ALTER TABLE waytone_erp_state ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow read-write for ERP anon client"
ON waytone_erp_state FOR ALL
TO anon, authenticated
USING (true) WITH CHECK (true);

-- 4. Enable Supabase Realtime broadcast
ALTER PUBLICATION supabase_realtime ADD TABLE waytone_erp_state;
